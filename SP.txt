
SET STATISTICS IO ON;
SET STATISTICS TIME ON;

DECLARE
    @IsEligible BIT,
    @ResultMessage NVARCHAR(500),
    @OutstandingAmount DECIMAL(18,2);

EXEC dbo.Check_Customer_Payment_Status
    @CompanyCode = '1049',
    @CustomerCode = 'YOUR_CUSTOMER_CODE',
    @Division = '25',
    @IsEligible = @IsEligible OUTPUT,
    @ResultMessage = @ResultMessage OUTPUT,
    @OutstandingAmount = @OutstandingAmount OUTPUT;

SELECT
    @IsEligible AS IsEligible,
    @ResultMessage AS ResultMessage,
    @OutstandingAmount AS OutstandingAmount;

SET STATISTICS IO OFF;
SET STATISTICS TIME OFF;
