EXEC sp_helpindex 'dbo.TB_CustomerAgeingReport';

-- 2. Total records
SELECT COUNT_BIG(*) AS TotalRecords
FROM dbo.TB_CustomerAgeingReport;

-- 3. Updated_by data type
SELECT
    COLUMN_NAME,
    DATA_TYPE
FROM INFORMATION_SCHEMA.COLUMNS
WHERE TABLE_SCHEMA = 'dbo'
  AND TABLE_NAME = 'TB_CustomerAgeingReport'
  AND COLUMN_NAME = 'updated_by';
