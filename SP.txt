
-- Check the database
SELECT DB_NAME() AS CurrentDatabase;

-- Check whether the index was created
SELECT
    index_id,
    name AS IndexName,
    type_desc AS IndexType
FROM sys.indexes
WHERE object_id =
    OBJECT_ID('dbo.TB_CustomerAgeingReport');

-- Check for paused/running resumable index operations
SELECT
    name,
    state_desc,
    percent_complete
FROM sys.index_resumable_operations
WHERE object_id =
    OBJECT_ID('dbo.TB_CustomerAgeingReport');

-- Check active requests
SELECT
    session_id,
    command,
    status,
    wait_type,
    blocking_session_id
FROM sys.dm_exec_requests
WHERE database_id = DB_ID()
  AND session_id <> @@SPID;

CREATE NONCLUSTERED INDEX
    IX_TB_CustomerAgeingReport_PaymentStatus
ON dbo.TB_CustomerAgeingReport
(
    CompanyCode,
    CustomerCode,
    CCA
)
WITH
(
    ONLINE = ON,
    RESUMABLE = ON,
    MAXDOP = 1,
    MAX_DURATION = 60 MINUTES
);
