SELECT
    index_id,
    name AS IndexName,
    type_desc AS IndexType
FROM sys.indexes
WHERE object_id = OBJECT_ID('dbo.TB_CustomerAgeingReport');





USE [DealerPortalDB];
GO

CREATE NONCLUSTERED INDEX
    IX_TB_CustomerAgeingReport_PaymentStatus
ON dbo.TB_CustomerAgeingReport
(
    CompanyCode,
    CustomerCode,
    CCA
)
INCLUDE
(
    GLAccount,
    updated_by
);
GO

SELECT TOP (10)
    updated_by,
    TRY_CONVERT(DATETIME, updated_by) AS ConvertedDate
FROM dbo.TB_CustomerAgeingReport
WHERE updated_by IS NOT NULL;




SELECT
    DB_NAME() AS CurrentDatabase,
    SERVERPROPERTY('Edition') AS Edition,
    DATABASEPROPERTYEX(DB_NAME(), 'ServiceObjective')
        AS ServiceObjective;

---------------------------
SELECT TOP (10)
    end_time,
    avg_cpu_percent,
    avg_data_io_percent,
    avg_log_write_percent,
    avg_memory_usage_percent
FROM sys.dm_db_resource_stats
ORDER BY end_time DESC;

-----------------------------
SELECT
    CAST(total_log_size_in_bytes / 1048576.0
        AS DECIMAL(18,2)) AS TotalLogMB,
    CAST(used_log_space_in_bytes / 1048576.0
        AS DECIMAL(18,2)) AS UsedLogMB,
    CAST(used_log_space_in_percent
        AS DECIMAL(18,2)) AS LogUsedPercent
FROM sys.dm_db_log_space_usage;
